%% Rechnungsvorlage "raster" – wird vom InvoiceRenderer gefüllt, mit denselben
%% Werten und Makroaufrufen wie "balken" (ADR-0011). Platzhalter, Makros und die
%% Regel, wer was entscheidet, beschreibt der Kopf von balken.tex. Kurz: Jede
%% einzelne Angabe darf leer sein, und diese Vorlage fragt dafür nur \ifblank.
%%
%% Gestaltungsprinzip: klassisch-tabellarisch und schwarz-weiß. Die Ordnung
%% trägt das Gitter der Positionen, nicht eine Farbe. Oben rechts der
%% Dokumenttitel mit den Kopfangaben, darunter Empfänger und Absender
%% nebeneinander, unten der Fuß in drei Spalten. Farbe bringt allenfalls das
%% Logo oben links mit, wenn die Stammdaten eines nennen.
\documentclass[10pt,a4paper]{article}

\setmainfont{FiraSans}[

  Extension      = .otf,

%% Keine Silbentrennung, aus demselben Grund wie in balken.tex: Eigennamen,

%% Gewöhnlicher Wortzwischenraum nach jedem Punkt (#143); den breiten
%% bekommen nur die eigenen Sätze des Programms in \hinweisabsatz.
\frenchspacing

%% Etwas Luft in den Zeilen des Gitters, sonst kleben die Ziffern an den Linien.
\renewcommand{\arraystretch}{1.35}

%% Die Breite der Betragsspalte. Positionen, Summen und Absetzung stehen in
%% drei Tabellen untereinander; nur mit derselben Breite fluchten ihre Linien.
%% Nachgemessen braucht "−19.040,00 €" rund 21 mm.
\newlength{\betragsbreite}\setlength{\betragsbreite}{24mm}
\newcolumntype{B}{>{\raggedleft\arraybackslash}p{\betragsbreite}}

%% Die Zeilen- und Tabellentrenner, die der Renderer aufruft (TemplateMacro).
%% Im Gitter trennt eine Linie die Positionen; die Summen stehen ohne Linie
%% untereinander, erst die Abschlusszeile setzt wieder eine.
\newcommand{\positionstrenner}{\\\hline}
\newcommand{\summentrenner}{\\}
\newcommand{\zeilenwechsel}{\\}
\newcommand{\zusammen}[1]{\mbox{#1}}

%% Eine Zeile oben rechts: Beschriftung und Wert als Tabellenzeile. Ist der Wert
%% leer, entfällt die ganze Zeile samt Beschriftung – sonst stünde auf einer
%% gewöhnlichen Rechnung "Vereinnahmt am" ohne Datum.
\newcommand{\kopfangabe}[2]{\ifblank{#2}{}{#1 & #2\\}}

%% Die eigenen Sätze des Programms unter den Summen, wie in balken.tex: nur hier
%% der breite Abstand nach einem Satzende (#143), deshalb keine Eingabedaten.
\newcommand{\hinweisabsatz}[1]{\par\vspace{5mm}\noindent%
  {\nonfrenchspacing\sfcode`\:=1000 \sfcode`\;=1000 \sfcode`\,=1000 #1}}

\newcommand{\absetzungskopf}[1]{\par\vspace{6mm}\noindent #1\par\vspace{1.5mm}}

%% Die abgesetzten Abschläge im selben Gitter wie die Summen: Bezeichnung links,
%% Betrag in der Betragsspalte. Der Renderer liefert die Zeilen ohne Zeilenende
%% nach der letzten, das setzt diese Tabelle.
\newcommand{\absetzungstabelle}[1]{\par\noindent%
  \begin{tabularx}{\linewidth}{|>{\raggedright\arraybackslash}X|B|}\hline#1\\\hline\end{tabularx}\par}

%% Die Zusatzzeile unter einer Position oder einem Abschlag. \normalfont, weil
%% die Bezeichnung einer Position fett steht und die Zeile darin liegt.
\newcommand{\positionsdetail}[1]{\newline{\normalfont\small #1}}
\newcommand{\absetzungsdetail}[1]{\newline{\small #1}}

%% Die Rabattspalte steht nur, wenn eine Position einen Rabatt hat (#213).
%% \mitrabatt setzt sein Argument nur dann, R ist die Spalte oder nichts.
\newcommand{\mitrabatt}[1]{\ifblank{<<RABATTSPALTE>>}{}{#1}}
\ifblank{<<RABATTSPALTE>>}{\newcolumntype{R}{}}{\newcolumntype{R}{>{\raggedleft\arraybackslash}p{14mm}|}}

%% Eine Zeile der Positionstabelle, Spalten wie das Gitter: laufende Nummer,
%% Bezeichnung (fett, die Beschreibung darunter), Menge, Einheit, Einzelpreis,
%% Rabatt, Gesamt.
\newcommand{\positionszeile}[7]{#1 & \textbf{#2} & #3 & #4 & #5\mitrabatt{ & #6} & #7}

\newcommand{\summenzeile}[2]{#1 & #2}

%% Die fette Zeile unter der Linie, mit der ein Betragsblock endet. Sie beendet
%% die Zeile davor selbst, wie in balken.tex (#147).
\newcommand{\abschlusszeile}[2]{\\\hline\textbf{#1} & \textbf{#2}}

%% Die Zeilen des Summenblocks mit ihren Beschriftungen (#217). Die Wörter
%% gehören zum Design; Kotlin liefert nur Satz und Beträge.
\newcommand{\nettozeile}[1]{\summenzeile{Nettobetrag}{#1}}
\newcommand{\steuerzeile}[2]{\summenzeile{zzgl. #1 USt.}{#2}}
\newcommand{\gesamtzeile}[1]{\abschlusszeile{Gesamtbetrag}{#1}}

%% Die Bankverbindung setzt raster nicht als eigenen Block: Sie steht im Fuß,
%% auf jedem Dokument, als Angabe des Briefbogens und nicht als
%% Zahlungsaufforderung – so, wie balken die IBAN in seiner Fußzeile führt.
%% Ob das Dokument zur Zahlung auffordert, sagt der Satz zur Zahlungsfrist.
%% Der Renderer ruft \bankblock trotzdem auf, wenn Zahlung gefordert wird, und
%% es bleibt leer (#189, ADR-0011).
\newcommand{\bankblock}{}

%% Das Logo nennen die Stammdaten (#198, ADR-0012). Der Renderer legt die
%% Datei neben die .tex, <<LOGO>> ist ihr Name, ohne Logo leer. Sie wird hier
%% in der Präambel gelesen und definiert (#199):
%%  - \logozeichnung, Pflicht: den Rumpf einer tikzpicture;
%%  - \logoskalierung, optional: einen Faktor auf \logohoehe, sonst 1.
%% Schrift und Farben bringt sie selbst mit; die Vorlage sichert zu, dass TikZ,
%% fontspec und xcolor geladen sind.
%%
%% Den Faktor begrenzt die Vorlage nicht, man sieht ihn am Blatt; die Grenze
%% steht beim Kopf.
%%
%% TikZ und keine Bilddatei: Der Treiber luatex.def von graphicx gibt jedem
%% über \includegraphics eingebundenen PDF eine Transparenzgruppe mit, und die
%% verbietet PDF/A-1 (ISO 19005-1, 6.4, siehe #197).
\newlength{\logohoehe}\setlength{\logohoehe}{12mm}
\newlength{\versalhoehe}
\ifblank{<<LOGO>>}{}{\input{<<LOGO>>}}
\providecommand{\logoskalierung}{1}
